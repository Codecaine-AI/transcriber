cmake_minimum_required(VERSION 3.20)
project(TranscriberTextEngine LANGUAGES C CXX)

if(NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Transcriber's server supports macOS and Linux.")
endif()

# Build separately from Whisper: its ggml version and targets are independent.
option(TRANSCRIBER_CUDA "Use CUDA for Linux inference" OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(APPLE)
    set(GGML_METAL ON CACHE BOOL "" FORCE)
    set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
    set(GGML_ACCELERATE ON CACHE BOOL "" FORCE)
    set(GGML_BLAS ON CACHE BOOL "" FORCE)
    set(GGML_BLAS_VENDOR Apple CACHE STRING "" FORCE)
    set(GGML_CUDA OFF CACHE BOOL "" FORCE)
else()
    set(GGML_METAL OFF CACHE BOOL "" FORCE)
    set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
    set(GGML_BLAS OFF CACHE BOOL "" FORCE)
    set(GGML_CUDA ${TRANSCRIBER_CUDA} CACHE BOOL "" FORCE)
    # Static cudart/cublas so the tarball does not need /usr/local/cuda.
    # libcuda.so.1 still comes from the host driver. Do not pull in NCCL.
    if(TRANSCRIBER_CUDA)
        set(GGML_STATIC ON CACHE BOOL "" FORCE)
        set(GGML_CUDA_NCCL OFF CACHE BOOL "" FORCE)
    endif()
endif()
find_package(Threads REQUIRED)

set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/llama.cpp")
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Run git submodule update --init --recursive first.")
endif()
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama")

add_executable(transcriber-text-engine worker.cpp)
target_compile_features(transcriber-text-engine PRIVATE cxx_std_17)
target_compile_options(transcriber-text-engine PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(transcriber-text-engine SYSTEM PRIVATE "${LLAMA_SOURCE}/vendor/nlohmann")
target_link_libraries(transcriber-text-engine PRIVATE llama Threads::Threads)
set_target_properties(transcriber-text-engine PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
